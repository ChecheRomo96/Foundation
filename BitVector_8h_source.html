61 // Makes `bytes` bytes available for the bits, filling new ones with
62 // `fill`. `false` and unchanged when the storage cannot hold them.
63 bool SetUsedBytes(size_t bytes, uint8_t fill) {
64 const size_t used = UsedBytes();
65 if (_external != nullptr) {
66 if (bytes > _externalBytes) {
67 return false;
68 }
69 for (size_t i = used; i < bytes; ++i) {
70 _external[i] = fill;
71 }
72 return true;
73 }
74 if (bytes == _owned.size()) {
75 return true;
76 }
77 if (bytes == _owned.size() + 1u) {
78 _owned.push_back(fill); // lets the vector grow its capacity geometrically
79 } else {
80 _owned.resize(bytes, fill);
81 }
82 return _owned.size() == bytes;
83 }

85 // Clears the bits from `first` to the end of the byte holding bit `first - 1`.
86 void ClearTail(size_t first) {
87 if (first % 8u != 0u) {
88 uint8_t& byte = Bytes()[first / 8u];
89 byte = static_cast<uint8_t>(byte & static_cast<uint8_t>(Mask(first) - 1u));
90 }
91 }

93 void Write(size_t index, bool value) {
94 uint8_t& byte = Bytes()[index / 8u];
95 byte = value ? static_cast<uint8_t>(byte | Mask(index))
96 : static_cast<uint8_t>(byte & static_cast<uint8_t>(~Mask(index)));
97 }

99 bool Read(size_t index) const {
100 return (static_cast<unsigned int>(Bytes()[index / 8u]) & Mask(index)) != 0u;
101 }

103 // Copies `other`'s bits into this storage; `false` and unchanged when they do not fit.
104 bool CopyBits(const BitVector& other) {
105 const size_t bytes = other.UsedBytes();
106 if (_external != nullptr) {
107 if (bytes > _externalBytes) {
108 return false;
109 }
110 for (size_t i = bytes; i < UsedBytes(); ++i) {
111 _external[i] = 0u; // bytes no longer in use
112 }
113 } else {
114 _owned.resize(bytes); // reuses the capacity when it is enough
115 if (_owned.size() != bytes) {
116 return false;
117 }
118 }
119 const uint8_t* source = other.Bytes();
120 uint8_t* destination = Bytes();
121 for (size_t i = 0; i < bytes; ++i) {
122 destination[i] = source[i];
123 }
124 _count = other._count;
125 return true;
126 }

263 bool Resize(size_t bits, bool value = false) {
264 const size_t previous = _count;
265 const size_t previousBytes = UsedBytes();
266 const size_t bytes = BytesFor(bits);
267 if (_external != nullptr && bytes > _externalBytes) {
268 return false;
269 }
270 if (!SetUsedBytes(bytes, value ? uint8_t(0xFFu) : uint8_t(0u))) {
271 return false;
272 }
273 if (_external != nullptr) {
274 for (size_t i = bytes; i < previousBytes; ++i) {
275 _external[i] = 0u;
276 }
277 }
278 _count = bits;
279 // Bits that were already stored in the previous last byte.
280 const size_t storedEnd = previousBytes * 8u;
281 for (size_t i = previous; i < bits && i < storedEnd; ++i) {
282 Write(i, value);
283 }
284 ClearTail(bits);
285 return true;
286 }

289 bool PushBack(bool value) {
290 if (_count == static_cast<size_t>(-1)) {
291 return false;
292 }
293 if (_count % 8u == 0u && !SetUsedBytes(UsedBytes() + 1u, uint8_t(0u))) {
294 return false;
295 }
296 Write(_count, value);
297 ++_count;
298 return true;
299 }
